import { existsSync, mkdtempSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
const { openStateDatabase } = await import(pathToFileURL(join(process.cwd(), 'src/state/connection.ts')).href);
const root = mkdtempSync(join(tmpdir(), 'lohra-sqlite-handle-'));
const connection = openStateDatabase(join(root, 'state.db'));
let beforeCode = 'none';
const openBefore = connection.database.open;
try { rmSync(root, { recursive: true, force: true }); }
catch (error) { beforeCode = error?.code ?? String(error); }
connection.close();
const openAfter = connection.database.open;
rmSync(root, { recursive: true, force: true });
const removed = !existsSync(root);
console.log(JSON.stringify({ node: process.version, platform: process.platform, arch: process.arch, openBefore, beforeCode, openAfter, removed }));
if (openBefore !== true || beforeCode !== 'EBUSY' || openAfter !== false || !removed) process.exitCode = 1;
